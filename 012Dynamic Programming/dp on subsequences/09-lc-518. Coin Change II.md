# Coin change II
problem link : https://leetcode.com/problems/coin-change-ii/

### Recursive:
```cpp
class Solution {
    int solve(int idx, int amt, vector<int>& coins) {
        // base case:
        if(idx == 0) {
            if(amt == 0) return 1;
            if(amt % coins[0] == 0) return 1;
            return 0;
        }

        int pick = 0;
        if(coins[idx] <= amt) pick = solve(idx, amt-coins[idx], coins);
        int notPick = solve(idx-1, amt, coins);

        return pick + notPick;
    }
public:
    int change(int amount, vector<int>& coins) {
        int n = coins.size();
        return solve(n-1, amount, coins);
    }
};
```

### Recursive + memoization:
```cpp
class Solution {
    int solve(int idx, int amt, vector<int>& coins,vector<vector<int>>& dp) {
        // base case:
        if(idx == 0) {
            if(amt == 0) return 1;
            if(amt % coins[0] == 0) return 1;
            return 0;
        }
        if(dp[idx][amt] != -1) return dp[idx][amt];
        int pick = 0;
        if(coins[idx] <= amt) pick = solve(idx, amt-coins[idx], coins, dp);
        int notPick = solve(idx-1, amt, coins, dp);

        return dp[idx][amt] = pick + notPick;
    }
public:
    int change(int amount, vector<int>& coins) {
        int n = coins.size();
        vector<vector<int>> dp(n, vector<int>(amount+1, -1));
        return solve(n-1, amount, coins, dp);
    }
};
```

### tabulation:
```cpp
class Solution {
public:
    int change(int amount, vector<int>& coins) {
        int n = coins.size();
        vector<vector<long long>> dp(n, vector<long long>(amount+1, 0));
        
        // base cases:
        dp[0][0] = 1;
        for(int amt = coins[0]; amt <= amount; amt += coins[0]){
            dp[0][amt] = 1;
        }

        for(int idx = 1; idx < n; idx++) {
            for(int amt = 0; amt <= amount; amt++) {
                int pick = 0;
                if(coins[idx] <= amt) pick = dp[idx][amt-coins[idx]];
                int notPick = dp[idx-1][amt];
                dp[idx][amt] = (long long)pick + notPick;
            }
        }
        return (int)dp[n-1][amount];
    }
};
```

### tabulation with space optimization:

```cpp
class Solution {
public:
    int change(int amount, vector<int>& coins) {
        int n = coins.size();
        // vector<vector<long long>> dp(n, vector<long long>(amount+1, 0));
        vector<long long> prev(amount+1, 0), curr(amount+1, 0);
                
        // base cases:
        prev[0] = 1;
        for(int amt = coins[0]; amt <= amount; amt += coins[0]){
            prev[amt] = 1;
        }

        for(int idx = 1; idx < n; idx++) {
            for(int amt = 0; amt <= amount; amt++) {
                int pick = 0;
                if(coins[idx] <= amt) pick = curr[amt-coins[idx]];
                int notPick = prev[amt];
                curr[amt] = (long long)pick + notPick;
            }
            prev = curr;
        }
        return (int)prev[amount];
    }
};
```
