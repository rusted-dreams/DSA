# 494. Target Sum
problem link: https://leetcode.com/problems/target-sum


let all no. having +, have sum s1.
all no. having -, have sum s2.
-> s1 - s2 = target.
sum of all no. = totalSum (s1 + s2 = total)
-> 2*s2 = total - target
-> s2 = (total - target)/2
hence resolved to the problem -> count subsets with sum k.

### Recursive:
```cpp
class Solution {
    int solve(int idx, int sum, vector<int>& nums) {
        // base case:
        if(idx == 0) {
            if(sum == 0 && nums[0] == 0) return 2;
            if(sum == 0 || nums[0] == sum) return 1;
            return 0;
        }

        int pick = 0;
        if(nums[idx] <= sum) pick = solve(idx-1, sum-nums[idx], nums);
        int notPick = solve(idx-1, sum, nums);

        return pick + notPick;
    }
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();
        int total = accumulate(nums.begin(), nums.end(), 0);
        int k = (total-target)/2;
        if(total-target <0) return 0;
        if((total-target)& 1) return 0;
        return solve(n-1, k, nums);
    }
};
```

### Recursive + memoization:
```cpp
class Solution {
    int solve(int idx, int sum, vector<int>& nums, vector<vector<int>>& dp) {
        // base case:
        if(idx == 0) {
            if(sum == 0 && nums[0] == 0) return 2;
            if(sum == 0 || nums[0] == sum) return 1;
            return 0;
        }
        if(dp[idx][sum] != -1) return dp[idx][sum];
        int pick = 0;
        if(nums[idx] <= sum) pick = solve(idx-1, sum-nums[idx], nums, dp);
        int notPick = solve(idx-1, sum, nums, dp);

        return dp[idx][sum] = pick + notPick;
    }
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();
        int total = accumulate(nums.begin(), nums.end(), 0);
        int k = (total-target)/2;
        if(total-target <0) return 0;
        if((total-target)& 1) return 0;
        vector<vector<int>> dp(n, vector<int>(k+1, -1));
        return solve(n-1, k, nums, dp);
    }
};
```

### Tabulation:
```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();
        int total = accumulate(nums.begin(), nums.end(), 0);
        int k = (total-target)/2;
        if(total-target <0) return 0;
        if((total-target)& 1) return 0;
        vector<vector<int>> dp(n, vector<int>(k+1, 0));
        // base cases:
        if(nums[0] == 0) dp[0][0] = 2;
        else dp[0][0] = 1;
        if(nums[0] != 0 && nums[0] <= k) dp[0][nums[0]] = 1;

        for(int idx = 1; idx < n; idx++) {
            for(int sum = 0; sum <= k; sum++) {
                int pick = 0;
                if(nums[idx] <= sum) pick = dp[idx-1][sum-nums[idx]];
                int notPick = dp[idx-1][sum];
                dp[idx][sum] = pick + notPick;
            }
        }
        return dp[n-1][k];
    }
};
```

### tabulation + space optimization:
```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();
        int total = accumulate(nums.begin(), nums.end(), 0);
        int k = (total-target)/2;
        if(total-target <0) return 0;
        if((total-target)& 1) return 0;
        // vector<vector<int>> dp(n, vector<int>(k+1, 0));

        vector<int> prev(k+1, 0), curr(k+1, 0);
        // base cases:
        if(nums[0] == 0) prev[0] = 2;
        else prev[0] = 1;
        if(nums[0] != 0 && nums[0] <= k) prev[nums[0]] = 1;

        for(int idx = 1; idx < n; idx++) {
            for(int sum = 0; sum <= k; sum++) {
                int pick = 0;
                if(nums[idx] <= sum) pick = prev[sum-nums[idx]];
                int notPick = prev[sum];
                curr[sum] = pick + notPick;
            }
            prev = curr;
        }
        return prev[k];
    }
};
```
