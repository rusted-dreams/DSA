# 322. Coin Change

problem link: https://leetcode.com/problems/coin-change

### Recursive:
```cpp
class Solution {
    int solve(int idx, int amt, vector<int>& coins) {
        if(idx == 0) {
            if(amt == 0) return 0;
            if(amt%coins[0] == 0) return amt/coins[0];
            return INT_MAX;
        }
        int notPick = solve(idx-1, amt, coins);
        int pick = INT_MAX;
        if(coins[idx] <= amt) 
            pick = 1 + solve(idx, amt-coins[idx], coins);
        return min(pick, notPick);
    }
public:
    int coinChange(vector<int>& coins, int amount) {
        int n = coins.size();
        int res = solve(n-1, amount, coins);
        return (res == INT_MAX)? -1 : res;
    }
};
```

### Recursive + Memoization:
```cpp
class Solution {
    int solve(int idx, int amt, vector<int>& coins, vector<vector<int>> &dp) {
        if(idx == 0) {
            if(amt == 0) return 0;
            if(amt%coins[0] == 0) return amt/coins[0];
            return INT_MAX;
        }
        if(dp[idx][amt] != -1) return dp[idx][amt];
        int notPick = solve(idx-1, amt, coins, dp);
        int pick = INT_MAX;
        if(coins[idx] <= amt) {
            pick = solve(idx, amt-coins[idx], coins, dp);
            if(pick < INT_MAX) pick++;
        }
        return dp[idx][amt] = min(pick, notPick);
    }
public:
    int coinChange(vector<int>& coins, int amount) {
        int n = coins.size();
        vector<vector<int>> dp(n, vector<int>(amount+1, -1));
        int res = solve(n-1, amount, coins, dp);
        return (res == INT_MAX)? -1 : res;
    }
};
```

```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        int n = coins.size();
        vector<vector<int>> dp(n, vector<int>(amount+1, 1e9));
        // base cases:
        for(int i = 0; i<=amount/coins[0]; i++) {
            dp[0][i*coins[0]] = i;
        }
        if(amount == 0) dp[0][0] = 0;
        
        for(int idx = 1; idx <n; idx++) {
            for(int amt = 0; amt <= amount; amt++) {
                int pick = 1e9;
                if(coins[idx] <= amt) pick = 1 + dp[idx][amt-coins[idx]];
                int notPick = dp[idx-1][amt];
                dp[idx][amt] = min(pick, notPick);
            }
        }
        return (dp[n-1][amount] == 1e9)? -1 : dp[n-1][amount];
    }
};
```

### tabulation + space optimization:

```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        int n = coins.size();
        vector<vector<int>> dp(n, vector<int>(amount+1, 1e9));
        vector<int> prev(amount+1, 1e9), curr(amount+1, 1e9);
        // base cases:
        for(int i = 0; i<=amount/coins[0]; i++) {
            prev[i*coins[0]] = i;
        }
        if(amount == 0) prev[0] = 0;
        
        for(int idx = 1; idx <n; idx++) {
            for(int amt = 0; amt <= amount; amt++) {
                int pick = 1e9;
                if(coins[idx] <= amt) pick = 1 + curr[amt-coins[idx]];
                int notPick = prev[amt];
                curr[amt] = min(pick, notPick);
            }
            prev = curr;
        }
        return (prev[amount] == 1e9)? -1 : prev[amount];

    }
};
```
